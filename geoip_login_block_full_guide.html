# prepare_csv.py — GeoLite2 Country CSV → Oracle 적재용 CSV 변환
# 출력: geoip_ranges.csv (NETWORK, START_IP_NUM, END_IP_NUM, COUNTRY_CODE, COUNTRY_NAME)
import csv, ipaddress

BLOCKS = 'GeoLite2-Country-Blocks-IPv4.csv'
LOCS   = 'GeoLite2-Country-Locations-en.csv'
OUT    = 'geoip_ranges.csv'

# 1) geoname_id → (국가코드, 국가명) 매핑
loc = {}
with open(LOCS, encoding='utf-8') as f:
    for r in csv.DictReader(f):
        loc[r['geoname_id']] = (r['country_iso_code'], r['country_name'])

# 2) 블록 CSV 변환
cnt = 0
with open(BLOCKS, encoding='utf-8') as f, \
     open(OUT, 'w', newline='', encoding='utf-8') as o:
    w = csv.writer(o)
    w.writerow(['NETWORK', 'START_IP_NUM', 'END_IP_NUM', 'COUNTRY_CODE', 'COUNTRY_NAME'])
    for r in csv.DictReader(f):
        # geoname_id 없으면 registered_country_geoname_id 로 대체
        gid = r['geoname_id'] or r['registered_country_geoname_id']
        if not gid or gid not in loc:
            continue
        net = ipaddress.ip_network(r['network'])
        cc, cn = loc[gid]
        w.writerow([r['network'],
                    int(net.network_address),
                    int(net.broadcast_address),
                    cc, cn])
        cnt += 1
print(f'완료: {cnt}건 → {OUT}')
참고 GeoLite2 Country IPv4 블록은 보통 30~50만 행입니다. 대역은 서로 겹치지 않으며 START_IP_NUM 기준 정렬 상태 — 이후 "내림차순 첫 행" 조회 최적화의 전제가 됩니다.

3Oracle 테이블 생성 (A/B 이중화 + 시노님)

재적재 중에도 서비스가 끊기지 않도록 물리 테이블 2개를 두고, 애플리케이션은 시노님 하나만 바라봅니다.

-- 물리 테이블 2개 (컬럼 구조 동일)
CREATE TABLE TB_CM_GEOIP_RANGE_A (
    NETWORK        VARCHAR2(30),
    START_IP_NUM   NUMBER,
    END_IP_NUM     NUMBER,
    COUNTRY_CODE   VARCHAR2(10),
    COUNTRY_NAME   VARCHAR2(100)
) TABLESPACE TS_NEO_DAT;

CREATE TABLE TB_CM_GEOIP_RANGE_B (
    NETWORK        VARCHAR2(30),
    START_IP_NUM   NUMBER,
    END_IP_NUM     NUMBER,
    COUNTRY_CODE   VARCHAR2(10),
    COUNTRY_NAME   VARCHAR2(100)
) TABLESPACE TS_NEO_DAT;

CREATE INDEX TB_CM_GEOIP_RANGE_A_X01 ON TB_CM_GEOIP_RANGE_A (START_IP_NUM, END_IP_NUM) TABLESPACE TS_NEO_IDX;
CREATE INDEX TB_CM_GEOIP_RANGE_B_X01 ON TB_CM_GEOIP_RANGE_B (START_IP_NUM, END_IP_NUM) TABLESPACE TS_NEO_IDX;

-- 애플리케이션/함수는 이 이름만 참조
CREATE OR REPLACE SYNONYM TB_CM_GEOIP_RANGE FOR TB_CM_GEOIP_RANGE_A;

5검증 · 통계수집 · 시노님 스위치

순서가 중요합니다: 적재 → 건수 검증 → 통계수집 → 스위치. 통계수집을 빠뜨리면 옵티마이저가 dynamic sampling에 의존해 조회가 수 초씩 걸릴 수 있습니다 (실제 겪은 이슈).

-- 1) 건수 검증 (신규가 기존의 90% 미만이면 데이터 이상으로 보고 중단)
DECLARE
    v_new_cnt NUMBER;
    v_old_cnt NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_new_cnt FROM TB_CM_GEOIP_RANGE_A;   -- 이번에 적재한 테이블
    SELECT COUNT(*) INTO v_old_cnt FROM TB_CM_GEOIP_RANGE_B;   -- 기존 테이블

    IF v_old_cnt > 0 AND v_new_cnt < v_old_cnt * 0.9 THEN
        RAISE_APPLICATION_ERROR(-20001, '신규 GeoIP 데이터 건수 이상 - 스위치 중단');
    END IF;
END;
/

6IP_TO_NUM — IP 문자열 → 숫자 변환 함수

CREATE OR REPLACE FUNCTION IP_TO_NUM(p_ip IN VARCHAR2) RETURN NUMBER IS
BEGIN
    RETURN
        TO_NUMBER(REGEXP_SUBSTR(p_ip, '[0-9]+', 1, 1)) * 16777216 +
        TO_NUMBER(REGEXP_SUBSTR(p_ip, '[0-9]+', 1, 2)) * 65536 +
        TO_NUMBER(REGEXP_SUBSTR(p_ip, '[0-9]+', 1, 3)) * 256 +
        TO_NUMBER(REGEXP_SUBSTR(p_ip, '[0-9]+', 1, 4));
EXCEPTION WHEN OTHERS THEN
    RETURN NULL;   -- IPv6·형식 오류 등은 NULL (호출측에서 매칭실패 처리)
END;
/

SELECT IP_TO_NUM('8.8.8.8') FROM DUAL;   -- 134744072 이면 정상

7IS_KOREA_IP — 한국 IP 여부 판정

성능 핵심 — "내림차순 첫 행" 패턴 BETWEEN START_IP_NUM AND END_IP_NUM 방식은 후보 구간을 거의 다 훑어야 해서 느립니다 (223.x대 IP에서 3초 발생 사례). GeoIP 대역은 겹치지 않으므로 "v 이하의 START_IP_NUM 중 최대값 1건"만 집어 END를 확인하는 방식으로 재작성 — 어떤 IP든 인덱스 접근이 사실상 1건으로 고정됩니다. 
CREATE OR REPLACE FUNCTION IS_KOREA_IP(
    p_ip IN VARCHAR2                 -- 체크할 IP 문자열
) RETURN VARCHAR2                    -- 'Y' = 한국, 'N' = 한국 아님(매칭 실패 포함)
IS
    v_country_code  VARCHAR2(10);
    v_num           NUMBER;
BEGIN
    v_num := IP_TO_NUM(p_ip);        -- 한 번만 변환해 상수화
    IF v_num IS NULL THEN
        RETURN 'N';                  -- 변환 실패(IPv6·형식오류)
    END IF;

    -- 내림차순 첫 행: v_num 이하 START 중 최대 1건 → 그 대역이 v_num 포함하는지 확인
    -- ※ end_ip_num 조건은 반드시 ROWNUM 바깥에 — 안에 넣으면 다시 전체 필터로 회귀
    SELECT country_code
    INTO   v_country_code
    FROM (
        SELECT country_code, end_ip_num
        FROM   TB_CM_GEOIP_RANGE
        WHERE  start_ip_num <= v_num
        ORDER BY start_ip_num DESC
    )
    WHERE ROWNUM = 1
    AND   end_ip_num >= v_num;

    RETURN CASE WHEN v_country_code = 'KR' THEN 'Y' ELSE 'N' END;
EXCEPTION
    WHEN NO_DATA_FOUND THEN RETURN 'N';   -- 사설 IP·미할당·대역 사이 빈 구간
    WHEN OTHERS       THEN RETURN 'N';
END;
/

SELECT IS_KOREA_IP('223.62.180.1') FROM DUAL;   -- 'Y'
SELECT IS_KOREA_IP('8.8.8.8')      FROM DUAL;   -- 'N'

8허용 테이블 TB_CM_IP_ALLOW (해외 예외 사용자)

해외거주자·해외법인·국외 출장자 등 예외 이용자를 승인 절차와 함께 관리합니다 (승인자·사유·기간 기록).

CREATE TABLE TB_CM_IP_ALLOW (

9IS_LOGIN_ALLOWED — 허용 여부 판정 함수

한국 IP이거나 허용테이블에 등재돼 있으면 'Y', 그 외 'N'. GeoIP 조회는 7장과 동일한 내림차순 방식입니다.

CREATE OR REPLACE FUNCTION IS_LOGIN_ALLOWED(
    p_user_id IN VARCHAR2,           -- 로그인 시도 사용자 ID
    p_ip      IN VARCHAR2            -- 로그인 시도 IP (실제 IP — XFF 추출 후)
) RETURN VARCHAR2                    -- 'Y' = 허용, 'N' = 차단
IS
    v_country_code  VARCHAR2(10);
    v_allow_cnt     NUMBER;
    v_num           NUMBER;
    v_now           VARCHAR2(14) := TO_CHAR(SYSDATE, 'YYYYMMDDHH24MISS');
BEGIN
    v_num := IP_TO_NUM(p_ip);

    -- 1) 한국 IP 확인 (내림차순 첫 행 방식)
    IF v_num IS NOT NULL THEN
        BEGIN
            SELECT country_code
            INTO   v_country_code
            FROM (
                SELECT country_code, end_ip_num
                FROM   TB_CM_GEOIP_RANGE
                WHERE  start_ip_num <= v_num
                ORDER BY start_ip_num DESC
            )
            WHERE ROWNUM = 1
            AND   end_ip_num >= v_num;
        EXCEPTION WHEN NO_DATA_FOUND THEN
            v_country_code := NULL;             -- 사설 IP / 미할당 등
        END;

        IF v_country_code = 'KR' THEN
            RETURN 'Y';
        END IF;
    END IF;

    -- 2) 한국이 아니면 허용 테이블 확인 (USER_ID + IP 정확 일치, 기간·사용여부 체크)
    SELECT COUNT(*)
    INTO   v_allow_cnt
    FROM   TB_CM_IP_ALLOW
    WHERE  USER_ID  = p_user_id
    AND    ALLOW_IP = p_ip
    AND    USE_YN   = 'Y'
    AND    (START_DT IS NULL OR v_now >= START_DT)
    AND    (END_DT   IS NULL OR v_now <= END_DT);

    RETURN CASE WHEN v_allow_cnt > 0 THEN 'Y' ELSE 'N' END;

EXCEPTION
    WHEN OTHERS THEN
        RETURN 'N';
END;
/

SET TIMING ON
SELECT IS_LOGIN_ALLOWED('hong123', '223.62.180.1') FROM DUAL;   -- 'Y' (즉시 응답이어야 정상)
SELECT IS_LOGIN_ALLOWED('hong123', '8.8.8.8')      FROM DUAL;   -- 'N'

10차단 이력 테이블 TB_WB_LOGIN_HIST_BLOCK

차단(또는 관찰 모드의 차단 대상) 건을 기록합니다. 컬럼 타입/길이는 TB_WB_LOGIN_HIST 실제 정의에 맞춰 조정하세요.

CREATE TABLE TB_WB_LOGIN_HIST_BLOCK
(
    PROC_DT                 VARCHAR2(  20),   -- 처리 일시 (YYYYMMDDHH24MISSFF)
    PROC_FG                 VARCHAR2(   1),   -- I(로그인) / C(메뉴) 등
    SEQ                     NUMBER,           -- 원본 TB_WB_LOGIN_HIST.SEQ

11공용 판정 함수 FN_CHK_LOGIN_BLOCK

판정 + 이력 기록 + 모드 스위치를 모두 담당하는 공용 함수입니다. 트리거·메뉴 함수 등 어디서든 호출하며, 차단 활성화 시 이 함수의 C_BLOCK_MODE 한 줄만 바꾸면 모든 호출처에 일괄 적용됩니다.

리턴값의미
'Y'허용 (SMS 미등록 / 한국 IP / 허용테이블 등재)
'M'차단 대상이지만 관찰 모드라 허용 (이력만 기록)
'N'차단 (이력 기록됨 — 호출측에서 RAISE 등 차단 처리)
 
CREATE OR REPLACE FUNCTION FN_CHK_LOGIN_BLOCK
/* ***********************************************************************************************
   국가 기반 로그인 제한 판정 (트리거·프로시저·화면 등 공용)
     1) SMS 사용등록 안 된 USER_ID          → 'Y' (검사 없이 허용)
     2) 한국 IP                            → 'Y'
     3) TB_CM_IP_ALLOW 등재                → 'Y'
     4) 그 외 = 차단 대상
        - 관찰 모드(C_BLOCK_MODE='N')       → 'M' (이력만 기록, 허용)
        - 차단 모드(C_BLOCK_MODE='Y')       → 'N' (이력 기록, 호출측에서 차단)
   ★ 차단 활성화 : C_BLOCK_MODE 'N'→'Y' 변경 후 함수만 재컴파일 (호출처 수정 불필요)
*********************************************************************************************** */
(
    p_proc_fg       IN VARCHAR2                     -- I/U/D/C (트리거 외 호출 시 'I' 권장)
,   p_seq           IN NUMBER       DEFAULT NULL
,   p_user_id       IN VARCHAR2                     -- 필수
,   p_login_ip      IN VARCHAR2                     -- 필수 (XFF 체인 원본 그대로)
,   p_stat_cd       IN VARCHAR2     DEFAULT NULL    -- ↓ 이하 이력 기록용
,   p_login_orgn    IN VARCHAR2     DEFAULT NULL
,   p_brwsr_info    IN VARCHAR2     DEFAULT NULL
,   p_login_date    IN VARCHAR2     DEFAULT NULL
,   p_login_dt      IN VARCHAR2     DEFAULT NULL
,   p_v_user_id     IN VARCHAR2     DEFAULT NULL
,   p_orgn_cd       IN VARCHAR2     DEFAULT NULL
,   p_session_id    IN VARCHAR2     DEFAULT NULL
,   p_log_yn        IN VARCHAR2     DEFAULT 'Y'     -- 차단 대상 시 이력 INSERT 여부
) RETURN VARCHAR2
IS
    -- ★★★ 차단 활성화 스위치 : 'N' 관찰(현재) / 'Y' 차단 ★★★
    C_BLOCK_MODE    CONSTANT VARCHAR2(1) := 'N';

    V_SMS_CNT       NUMBER;
    V_REAL_IP       VARCHAR2(  45);
    V_IP_NUM        NUMBER;
    V_COUNTRY_CODE  VARCHAR2(  10);
    V_ALLOW_YN      VARCHAR2(   1);
    V_BLOCK_REASON  VARCHAR2( 200);

    -- 차단 이력 INSERT (AUTONOMOUS — 호출측 롤백과 무관하게 커밋)
    PROCEDURE SUB_SP_INSERT_BLOCK_HIST
    IS
        PRAGMA AUTONOMOUS_TRANSACTION;
    BEGIN
        INSERT INTO TB_WB_LOGIN_HIST_BLOCK
        (
                PROC_DT, PROC_FG, SEQ, USER_ID, STAT_CD, LOGIN_ORGN
        ,       LOGIN_IP, REAL_IP, COUNTRY_CODE, BLOCK_REASON
        ,       BRWSR_INFO, LOGIN_DATE, LOGIN_DT, V_USER_ID, ORGN_CD, SESSION_ID
        ,       USERENV_SESSIONID, USERENV_OS_USER, USERENV_SESSION_USER
        ,       USERENV_MODULE, USERENV_HOST, USERENV_DB_NAME
        ,       USERENV_INSTANCE_NAME, USERENV_SID, USERENV_TERMINAL, USERENV_IP_ADDRESS
        )
        VALUES
        (
                TO_CHAR(SYSTIMESTAMP, 'YYYYMMDDHH24MISSFF')
        ,       p_proc_fg, p_seq, p_user_id, p_stat_cd, p_login_orgn
        ,       p_login_ip, V_REAL_IP, V_COUNTRY_CODE, V_BLOCK_REASON
        ,       p_brwsr_info, p_login_date, p_login_dt, p_v_user_id, p_orgn_cd, p_session_id
        ,       SYS_CONTEXT('USERENV','SESSIONID'),  SYS_CONTEXT('USERENV','OS_USER')
        ,       SYS_CONTEXT('USERENV','SESSION_USER'), SYS_CONTEXT('USERENV','MODULE')
        ,       SYS_CONTEXT('USERENV','HOST'), SYS_CONTEXT('USERENV','DB_NAME')
        ,       SYS_CONTEXT('USERENV','INSTANCE_NAME'), SYS_CONTEXT('USERENV','SID')
        ,       SYS_CONTEXT('USERENV','TERMINAL'), SYS_CONTEXT('USERENV','IP_ADDRESS')
        );
        COMMIT;
    END SUB_SP_INSERT_BLOCK_HIST;

BEGIN
    -- 1) 제한 대상 확인 : SMS 사용등록 한 USER_ID 만 검사
    SELECT  COUNT(*)
    INTO    V_SMS_CNT
    FROM    TB_CM_SMS_USER_REGIST A
    WHERE   A.USER_ID   =   p_user_id
    AND     ROWNUM      =   1;

    IF V_SMS_CNT = 0 THEN
        RETURN 'Y';
    END IF;

    -- 2) 실제 클라이언트 IP 추출 (XFF 체인 대응 — 콤마 있으면 마지막 토큰)
    V_REAL_IP := TRIM(p_login_ip);
    IF INSTR(V_REAL_IP, ',') > 0 THEN
        V_REAL_IP := TRIM(SUBSTR(V_REAL_IP, INSTR(V_REAL_IP, ',', -1) + 1));
    END IF;

    -- 3) 허용 여부 판정
    V_ALLOW_YN := IS_LOGIN_ALLOWED(p_user_id, V_REAL_IP);
    IF V_ALLOW_YN = 'Y' THEN
        RETURN 'Y';
    END IF;

    -- 4) 차단 대상 : 국가코드 조회(이력용) → 이력 INSERT
    V_IP_NUM := IP_TO_NUM(V_REAL_IP);
    IF V_IP_NUM IS NOT NULL THEN
        BEGIN
            SELECT  COUNTRY_CODE
            INTO    V_COUNTRY_CODE
            FROM    (
                        SELECT  COUNTRY_CODE, END_IP_NUM
                        FROM    TB_CM_GEOIP_RANGE
                        WHERE   START_IP_NUM <= V_IP_NUM
                        ORDER   BY START_IP_NUM DESC
                    )
            WHERE   ROWNUM = 1
            AND     END_IP_NUM >= V_IP_NUM;
        EXCEPTION WHEN NO_DATA_FOUND THEN
            V_COUNTRY_CODE := NULL;
        END;
    END IF;

    V_BLOCK_REASON := CASE WHEN C_BLOCK_MODE = 'Y' THEN 'BLOCKED' ELSE 'MONITOR' END
                      || ' - Not Korea IP / Not Allowed'
                      || ' [CC=' || NVL(V_COUNTRY_CODE, 'UNKNOWN') || ']';

    IF p_log_yn = 'Y' THEN
        SUB_SP_INSERT_BLOCK_HIST;
    END IF;

    -- 5) 모드에 따른 최종 리턴
    RETURN CASE WHEN C_BLOCK_MODE = 'Y' THEN 'N' ELSE 'M' END;

EXCEPTION
    WHEN OTHERS THEN
        RETURN 'Y';   -- 판정 오류로 전체 로그인이 막히는 사고 방지 (허용 우선)
                      -- 차단 전환 후 보안 우선 정책이면 'N' 으로 변경
END;
/

12로그인 트리거 TR_TB_WB_LOGIN_HIST_R90

트리거는 함수 호출 + 결과 'N'일 때 RAISE만 담당하는 얇은 구조입니다.

CREATE OR REPLACE
TRIGGER SBPORA.TR_TB_WB_LOGIN_HIST_R90
/* ***********************************************************************************************
   국가 기반 로그인 제한 — 판정은 FN_CHK_LOGIN_BLOCK 위임
     'Y' 허용 / 'M' 관찰모드 허용(이력만) / 'N' 차단(RAISE)
   차단 활성화 : FN_CHK_LOGIN_BLOCK 의 C_BLOCK_MODE 'N'→'Y' (트리거 수정 불필요)
*********************************************************************************************** */
       AFTER INSERT ON "SBPORA"."TB_WB_LOGIN_HIST" FOR EACH ROW
DECLARE
    V_TRG_NM        VARCHAR2(  50) := 'TR_TB_WB_LOGIN_HIST_R90';
    V_USER_DEF_EXP  EXCEPTION;
    V_ERR_MSG       VARCHAR2(1000);
    V_LINE_NM       VARCHAR2(   4);
    V_PROC_FG       VARCHAR2(   1);
    V_CHK_RESULT    VARCHAR2(   1);
BEGIN
    IF INSERTING THEN
        V_PROC_FG := 'I';
        V_LINE_NM := 'CHK';

        V_CHK_RESULT := FN_CHK_LOGIN_BLOCK(
                            p_proc_fg       =>  V_PROC_FG
                        ,   p_seq           =>  :new.SEQ
                        ,   p_user_id       =>  :new.USER_ID
                        ,   p_login_ip      =>  :new.LOGIN_IP
                        ,   p_stat_cd       =>  :new.STAT_CD
                        ,   p_login_orgn    =>  :new.LOGIN_ORGN
                        ,   p_brwsr_info    =>  :new.BRWSR_INFO
                        ,   p_login_date    =>  :new.LOGIN_DATE
                        ,   p_login_dt      =>  :new.LOGIN_DT
                        ,   p_v_user_id     =>  :new.V_USER_ID
                        ,   p_orgn_cd       =>  :new.ORGN_CD
                        ,   p_session_id    =>  :new.SESSION_ID
                        );

        -- 'N' 일 때만 차단. 'Y'·'M' 은 통과
        IF V_CHK_RESULT = 'N' THEN
            V_ERR_MSG := 'Login Blocked: Country Restriction'
                         ||','|| :new.USER_ID
                         ||','|| :new.LOGIN_IP
                         ||','|| :new.STAT_CD
                         ||','|| :new.SESSION_ID
                         ||','|| TO_CHAR(SYSTIMESTAMP, 'YYYYMMDDHH24MISS');
            RAISE V_USER_DEF_EXP;
        END IF;
    END IF;

EXCEPTION
    WHEN V_USER_DEF_EXP THEN
        V_ERR_MSG := '[' || V_TRG_NM || ' : ' || V_LINE_NM || ']-' ||'['||V_ERR_MSG||']-'|| SQLERRM;
        RAISE_APPLICATION_ERROR(-20001, V_ERR_MSG);
    WHEN OTHERS THEN
        V_ERR_MSG := '[' || V_TRG_NM || ' : ' || V_LINE_NM || ']-' ||'['||V_ERR_MSG||']-'|| SQLERRM;
        RAISE_APPLICATION_ERROR(-20002, V_ERR_MSG);
END;
/

14운영 — 관찰 모드 → 차단 전환

14-1. 관찰 기간 (권장 1~2주 이상)

C_BLOCK_MODE = 'N' 상태에서는 아무도 차단되지 않고 TB_WB_LOGIN_HIST_BLOCK에 MONITOR - ... 이력만 쌓입니다. 주기적으로 오탐을 점검합니다.

-- 차단 예정 건 집계 (오탐 점검)
SELECT  USER_ID
,       REAL_IP
,       COUNTRY_CODE
,       COUNT(*)            AS CNT
,       MIN(LOGIN_DATE)     AS FIRST_TRY
,       MAX(LOGIN_DATE)     AS LAST_TRY
FROM    TB_WB_LOGIN_HIST_BLOCK
GROUP
BY      USER_ID, REAL_IP, COUNTRY_CODE
ORDER
BY      CNT DESC
;

-- 사설 IP·매칭 실패 건만 (내부망 유입 오탐 후보 — 반드시 확인)
SELECT  USER_ID, REAL_IP, COUNT(*) AS CNT
FROM    TB_WB_LOGIN_HIST_BLOCK
WHERE   COUNTRY_CODE IS NULL
GROUP   BY USER_ID, REAL_IP
ORDER   BY CNT DESC
;
  • 정상 사용자(출장자·해외법인)는 TB_CM_IP_ALLOW에 승인 절차와 함께 등록
  • 사설 IP 유입(내부망 이슈)이 확인되면 허용테이블 등록 또는 사설 대역(10.x, 172.16~31.x, 192.168.x) 예외 정책 결정

15운영 — 월간 GeoIP 재적재 절차 (A/B 무중단 스위칭)

GeoLite2는 주기적으로 갱신됩니다(주 2회 배포, 월 1회 재적재 권장). 현재 시노님이 A를 가리키면 B에 적재하고, B를 가리키면 A에 적재합니다.

-- 0) 현재 시노님 대상 확인 → 반대쪽이 "대기 테이블"
SELECT table_name FROM user_synonyms WHERE synonym_name = 'TB_CM_GEOIP_RANGE';

-- 1) 최신 zip 다운로드 → 압축해제 → prepare_csv.py 실행 (2장과 동일)

-- 2) 대기 테이블 비우고 적재 (예: 현재 A 사용 중 → B에 적재)
TRUNCATE TABLE TB_CM_GEOIP_RANGE_B;
--   sqlldr USERID=... CONTROL=geoipB.ctl SKIP=1

-- 3) 건수 검증 (신규 < 기존×0.9 이면 중단)
DECLARE
    v_new_cnt NUMBER;
    v_old_cnt NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_new_cnt FROM TB_CM_GEOIP_RANGE_B;
    SELECT COUNT(*) INTO v_old_cnt FROM TB_CM_GEOIP_RANGE_A;
    IF v_new_cnt < v_old_cnt * 0.9 THEN
        RAISE_APPLICATION_ERROR(-20001, '신규 GeoIP 데이터 건수 이상 - 스위치 중단');
    END IF;
END;
/

-- 4) 통계수집 (★ 재적재 때마다 필수 — 누락 시 조회 성능 급락)
EXEC DBMS_STATS.GATHER_TABLE_STATS(USER, 'TB_CM_GEOIP_RANGE_B', CASCADE => TRUE);

-- 5) 검증 통과 후에만 스위치 (원자적/즉시 적용 — 무중단)
CREATE OR REPLACE SYNONYM TB_CM_GEOIP_RANGE FOR TB_CM_GEOIP_RANGE_B;

-- 6) 스위치 후 동작 확인
SELECT COUNT(*) FROM TB_CM_GEOIP_RANGE;
